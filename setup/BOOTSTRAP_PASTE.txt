name: Bootstrap
on: {workflow_dispatch}
permissions: {contents: write}
jobs:
  unpack:
    runs-on: ubuntu-latest
    timeout-minutes: 30
    steps:
      - uses: actions/checkout@v4
      - name: download seed release
        env: {GH_TOKEN: "${{ github.token }}"}
        run: gh release download seed --pattern "*.zip" --repo "$GITHUB_REPOSITORY" -O seed.zip
      - name: unpack
        run: unzip -o seed.zip -x ".github/*" && rm seed.zip
      - name: commit tree
        run: |
          git config user.name "bootstrap"; git config user.email "bot@none"
          git add -A
          git commit -m "bootstrap: unpack seed [skip ci]" || echo "nothing to commit"
          git push
