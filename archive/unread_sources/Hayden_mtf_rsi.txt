// RSI CALCULATION

adjRsi(float v, simple int l) =>
    f = -math.pow(math.abs(math.abs(v - 50) - 50),
          1 + math.pow(l / 14.0, 0.618) - 1) /
          math.pow(50, math.pow(l / 14.0, 0.618) - 1) + 50
    v > 50 ? f + 50 : -f + 50

a    = 1.0 / len
u    = math.max(src - src[1], 0)
d    = math.max(src[1] - src,  0)
ruh  = a * math.max(high - close[1], 0) + (1 - a) * ta.rma(u, len)[1]
rdh  = (1 - a) * ta.rma(d, len)[1]
rul  = (1 - a) * ta.rma(u, len)[1]
rdl  = a * math.max(close[1] - low, 0) + (1 - a) * ta.rma(d, len)[1]

rsiha = 100 - 100 / (1 + ruh / rdh)
rsila = 100 - 100 / (1 + rul / rdl)
rsia  = ta.rsi(src, len)

rsih = ad ? adjRsi(rsiha, len) : rsiha
rsil = ad ? adjRsi(rsila, len) : rsila
rsi  = ad ? adjRsi(rsia,  len) : rsia

// ── Combined MTF function: state machine + divergence detection ───
f_mtf() =>
    rsi_v = ad ? adjRsi(ta.rsi(src, len), len) : ta.rsi(src, len)

    // State machine
    var int s = 0
    if ta.crossover(rsi_v, ts1)
        s := 1
    if ta.crossunder(rsi_v, ts2)
        s := 2
    if s == 1 and ta.crossunder(rsi_v, ts2a)
        s := 3
    if s == 2 and ta.crossover(rsi_v, ts1a)
        s := 3

    // Running extremes + prior archives
    var float runBearLow_m  = na
    var float runBearRsi_m  = na
    var array<float> priorBearLows_m = array.new_float()
    var array<float> priorBearRsis_m = array.new_float()

    var float runBullHigh_m = na
    var float runBullRsi_m  = na
    var array<float> priorBullHighs_m = array.new_float()
    var array<float> priorBullRsis_m  = array.new_float()

    if s == 2
        if na(runBearLow_m) or low < runBearLow_m
            runBearLow_m := low
            runBearRsi_m := rsi_v
    if s == 1
        if na(runBullHigh_m) or high > runBullHigh_m
            runBullHigh_m := high
            runBullRsi_m  := rsi_v

    // Divergence detection — current run extreme vs prior archived states
    int divType = 0

    if s == 2 and not na(runBearLow_m) and array.size(priorBearLows_m) > 0
        scanN_m = math.min(array.size(priorBearLows_m), mtfDivLookback)
        for i = 0 to scanN_m - 1
            if divType == 0
                pLow = array.get(priorBearLows_m, i)
                pRsi = array.get(priorBearRsis_m, i)
                if runBearLow_m < pLow and runBearRsi_m > pRsi
                    divType := 1   // RD bull
                else if runBearLow_m > pLow and runBearRsi_m < pRsi
                    divType := 2   // HD bull

    if s == 1 and not na(runBullHigh_m) and array.size(priorBullHighs_m) > 0
        scanN_m = math.min(array.size(priorBullHighs_m), mtfDivLookback)
        for i = 0 to scanN_m - 1
            if divType == 0
                pHigh = array.get(priorBullHighs_m, i)
                pRsi  = array.get(priorBullRsis_m,  i)
                if runBullHigh_m > pHigh and runBullRsi_m < pRsi
                    divType := 3   // RD bear
                else if runBullHigh_m < pHigh and runBullRsi_m > pRsi
                    divType := 4   // HD bear

    // Archive on leaving state
    leavingBear_m = s != 2 and s[1] == 2
    leavingBull_m = s != 1 and s[1] == 1

    if leavingBear_m and not na(runBearLow_m)
        array.unshift(priorBearLows_m, runBearLow_m)
        array.unshift(priorBearRsis_m, runBearRsi_m)
        while array.size(priorBearLows_m) > mtfDivLookback
            array.pop(priorBearLows_m)
            array.pop(priorBearRsis_m)
        runBearLow_m := na
        runBearRsi_m := na

    if leavingBull_m and not na(runBullHigh_m)
        array.unshift(priorBullHighs_m, runBullHigh_m)
        array.unshift(priorBullRsis_m,  runBullRsi_m)
        while array.size(priorBullHighs_m) > mtfDivLookback
            array.pop(priorBullHighs_m)
            array.pop(priorBullRsis_m)
        runBullHigh_m := na
        runBullRsi_m  := na

    [s[1], divType[1]]

[state_tf1, divType_tf1] = request.security(syminfo.tickerid, mtfTf1, f_mtf(), lookahead=barmerge.lookahead_on)

// ENGULFING — TIERED L/M/H SIGNALS (with body buffer)

greng = "🕯 Engulfing Signals 🕯"
showEng         = input.bool(true,  "Show engulfing signals?",          group=greng)
requireOppCandle= input.bool(true,  "Require opposite price candle?",   group=greng)
engBufPct       = input.float(1.0,  "Body buffer (% of prev body)",
  minval=0, maxval=20, step=0.1,                                         group=greng)
wickPctEng      = input.float(50.0, "Max total wick % of body",
  minval=0, maxval=500, step=5,                                          group=greng)
slopeLenEng     = input.int(40,     "Slope lookback (bars)",
  minval=1, maxval=500,                                                  group=greng)
compLen         = input.int(20,     "Prior move lookback (bars)",
  minval=5, maxval=200,                                                  group=greng)

p1_H = input.float(10.0, "H: body ≥",   minval=0, group=grP1)
p1_M = input.float(5.0,  "M: body ≥",   minval=0, group=grP1)

p2_bull_H = input.float(20.0, "Bull H: RSI open ≤", group=grP2)
p2_bull_M = input.float(30.0, "Bull M: RSI open ≤", group=grP2)
p2_bull_L = input.float(40.0, "Bull L: RSI open ≤", group=grP2)
p2_bear_H = input.float(80.0, "Bear H: RSI open ≥", group=grP2)
p2_bear_M = input.float(70.0, "Bear M: RSI open ≥", group=grP2)
p2_bear_L = input.float(60.0, "Bear L: RSI open ≥", group=grP2)

engColorBear = input.color(#ff1100, "Bear engulf color", group=grEngVis)

engFlashBear = input.color(color.yellow, "Bear flash bar color", group=grEngVis)

rsiC  = rsi
rsiH  = rsih
rsiL  = rsil
rsiO1 = rsi[2]
rsiC1 = rsi[1]

prevBodyHi  = math.max(rsiO1, rsiC1)
prevBodyLo  = math.min(rsiO1, rsiC1)
prevBodyAbs = math.abs(rsiC1 - rsiO1)
engBuf      = (engBufPct / 100.0) * prevBodyAbs

bullEngulf = rsiC > rsiO and rsiC1 < rsiO1 and
             rsiO <= prevBodyLo + engBuf and rsiC >= prevBodyHi - engBuf

bearEngulf = rsiC < rsiO and rsiC1 > rsiO1 and
             rsiO >= prevBodyHi - engBuf and rsiC <= prevBodyLo + engBuf

rsiBody    = math.abs(rsiC - rsiO)
upperWickR = rsiH - math.max(rsiO, rsiC)
lowerWickR = math.min(rsiO, rsiC) - rsiL
totalWickR = upperWickR + lowerWickR
wickOk     = rsiBody > 0 and totalWickR < (wickPctEng / 100.0) * rsiBody

rsiSlope    = rsi - rsi[slopeLenEng]
bullSlopeOk = rsiSlope < 0
bearSlopeOk = rsiSlope > 0

bullStateOk = state == 2
bearStateOk = state == 1
bullPriceOk = requireOppCandle ? close < open : true
bearPriceOk = requireOppCandle ? close > open : true

bullRaw = bullEngulf and wickOk and bullStateOk and bullSlopeOk and bullPriceOk
bearRaw = bearEngulf and wickOk and bearStateOk and bearSlopeOk and bearPriceOk

tier1(float bodySize) =>
    bodySize >= p1_H ? 3 : bodySize >= p1_M ? 2 : 1

tier2_bull(float rsiOpenVal) =>
    rsiOpenVal <= p2_bull_H ? 3 :
     rsiOpenVal <= p2_bull_M ? 2 :
     rsiOpenVal <= p2_bull_L ? 1 : 0

tier2_bear(float rsiOpenVal) =>
    rsiOpenVal >= p2_bear_H ? 3 :
     rsiOpenVal >= p2_bear_M ? 2 :
     rsiOpenVal >= p2_bear_L ? 1 : 0

tier3(float intensity) =>
    intensity <= p3_H ? 3 : intensity <= p3_M ? 2 : 1

priorRsiHi = ta.highest(rsi, compLen)[1]
priorRsiLo = ta.lowest(rsi,  compLen)[1]

bullIntensity = priorRsiHi - rsiL
bearIntensity = rsiH - priorRsiLo

bullT1 = bullRaw ? tier1(rsiBody)         : 0
bullT2 = bullRaw ? tier2_bull(rsiO)       : 0
bullT3 = bullRaw ? tier3(bullIntensity)   : 0
bullGrade = bullRaw and bullT1 > 0 and bullT2 > 0 and bullT3 > 0 ?
             math.min(bullT1, math.min(bullT2, bullT3)) : 0

bearT1 = bearRaw ? tier1(rsiBody)         : 0
bearT2 = bearRaw ? tier2_bear(rsiO)       : 0
bearT3 = bearRaw ? tier3(bearIntensity)   : 0
bearGrade = bearRaw and bearT1 > 0 and bearT2 > 0 and bearT3 > 0 ?
             math.min(bearT1, math.min(bearT2, bearT3)) : 0

gradeStr(int g) =>
    g == 3 ? "H" : g == 2 ? "M" : g == 1 ? "L" : ""
