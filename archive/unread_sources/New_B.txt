// profile -> scalars (POC/VAH/VAL/HVN/LVN), outlier-clipped. Computes EVERY bar (needed for historical forecast backfill).
f_levels(simple int nb, float vap, simple int len, float kSig) =>
    int n = len
    float poc = na
    float vah = na
    float val = na
    float hvn1 = na
    float lvn1 = na
    float amin = na
    float amax = na
    float sum = 0.0
    float ssq = 0.0
    int cnt = 0
    for k = 0 to n - 1
        float hk = high[k]
        float lk = low[k]
        float ck = close[k]
        if not na(hk) and not na(lk) and not na(ck)
            amin := na(amin) ? lk : math.min(amin, lk)
            amax := na(amax) ? hk : math.max(amax, hk)
            float src = (hk + lk + ck) / 3.0
            sum := sum + src
            ssq := ssq + src * src
            cnt := cnt + 1
    if cnt > 0 and not na(amin) and not na(amax) and amax > amin
        float mean = sum / cnt
        float vrc = math.max(0.0, ssq / cnt - mean * mean)
        float sd = math.sqrt(vrc)
        float loB = math.max(amin, mean - kSig * sd)
        float hiB = math.min(amax, mean + kSig * sd)
        if hiB <= loB
            loB := amin
            hiB := amax
        float bs = (hiB - loB) / nb
        arrV = array.new_float(nb, 0.0)
        for k = 0 to n - 1
            float hk = high[k]
            float lk = low[k]
            float ck = close[k]
            float vk = volume[k]
            if not na(hk) and not na(lk) and not na(ck) and not na(vk)
                float src = (hk + lk + ck) / 3.0
                if src >= loB and src <= hiB
                    int bi = int((src - loB) / bs)
                    bi := bi < 0 ? 0 : (bi > nb - 1 ? nb - 1 : bi)
                    array.set(arrV, bi, array.get(arrV, bi) + vk)
        int pocBin = 0
        float pocVol = -1.0
        float tot = 0.0
        for i = 0 to nb - 1
            float v = array.get(arrV, i)
            tot := tot + v
            if v > pocVol
                pocVol := v
                pocBin := i
        poc := loB + (pocBin + 0.5) * bs
        float meanV = tot / nb
        float target = tot * vap
        float acc = pocVol
        int lo = pocBin
        int hi = pocBin
        while acc < target and (lo > 0 or hi < nb - 1)
            float loV = lo > 0 ? array.get(arrV, lo - 1) : -1.0
            float hiV = hi < nb - 1 ? array.get(arrV, hi + 1) : -1.0
            if hiV >= loV
                hi := hi + 1
                acc := acc + (hiV > 0 ? hiV : 0.0)
            else
                lo := lo - 1
                acc := acc + (loV > 0 ? loV : 0.0)
        vah := loB + (hi + 1) * bs
        val := loB + lo * bs
        float h1v = -1.0
        int   h1b = -1
        float l1v = 1e18
        int   l1b = -1
        for i = 1 to nb - 2
            float v  = array.get(arrV, i)
            float vm = array.get(arrV, i - 1)
            float vp = array.get(arrV, i + 1)
            if v >= vm and v >= vp and i != pocBin and v > h1v
                h1v := v
                h1b := i
            if v <= vm and v <= vp and v < l1v
                l1v := v
                l1b := i
        hvn1 := h1b >= 0 ? loB + (h1b + 0.5) * bs : na
        lvn1 := l1b >= 0 ? loB + (l1b + 0.5) * bs : na
    [poc, vah, val, hvn1, lvn1]

// volume-at-price geometry around current price (for the STRUCTURE read). Computes every bar.
f_geom(simple int nb, simple int len, float kSig, float hvnM, float lvnM) =>
    int n = len
    float hvnUp = na
    float hvnDn = na
    float lvnUp = na
    float lvnDn = na
    int atCode = 1
    float amin = na
    float amax = na
    float sum = 0.0
    float ssq = 0.0
    int cnt = 0
    for k = 0 to n - 1
        float hk = high[k]
        float lk = low[k]
        float ck = close[k]
        if not na(hk) and not na(lk) and not na(ck)
            amin := na(amin) ? lk : math.min(amin, lk)
            amax := na(amax) ? hk : math.max(amax, hk)
            float src = (hk + lk + ck) / 3.0
            sum := sum + src
            ssq := ssq + src * src
            cnt := cnt + 1
    if cnt > 0 and not na(amin) and not na(amax) and amax > amin
        float mean = sum / cnt
        float vrc = math.max(0.0, ssq / cnt - mean * mean)
        float sd = math.sqrt(vrc)
        float loB = math.max(amin, mean - kSig * sd)
        float hiB = math.min(amax, mean + kSig * sd)
        if hiB <= loB
            loB := amin
            hiB := amax
        float bs = (hiB - loB) / nb
        arrV = array.new_float(nb, 0.0)
        for k = 0 to n - 1
            float hk = high[k]
            float lk = low[k]
            float ck = close[k]
            float vk = volume[k]
            if not na(hk) and not na(lk) and not na(ck) and not na(vk)
                float src = (hk + lk + ck) / 3.0
                if src >= loB and src <= hiB
                    int bi = int((src - loB) / bs)
                    bi := bi < 0 ? 0 : (bi > nb - 1 ? nb - 1 : bi)
                    array.set(arrV, bi, array.get(arrV, bi) + vk)
        float tot = 0.0
        for i = 0 to nb - 1
            tot := tot + array.get(arrV, i)
        float meanV = tot / nb
        float pr = close
        int pBin = int((pr - loB) / bs)
        pBin := pBin < 0 ? 0 : (pBin > nb - 1 ? nb - 1 : pBin)
        float atVol = array.get(arrV, pBin)
        atCode := atVol > meanV * hvnM ? 2 : atVol < meanV * lvnM ? 0 : 1
        for i = 1 to nb - 2
            float v  = array.get(arrV, i)
            float vm = array.get(arrV, i - 1)
            float vp = array.get(arrV, i + 1)
            float bp = loB + (i + 0.5) * bs
            bool isMax = v >= vm and v >= vp and v > meanV * hvnM
            bool isMin = v <= vm and v <= vp and v < meanV * lvnM
            if isMax
                if bp > pr
                    hvnUp := na(hvnUp) ? bp : math.min(hvnUp, bp)
                if bp < pr
                    hvnDn := na(hvnDn) ? bp : math.max(hvnDn, bp)
            if isMin
                if bp > pr
                    lvnUp := na(lvnUp) ? bp : math.min(lvnUp, bp)
                if bp < pr
                    lvnDn := na(lvnDn) ? bp : math.max(lvnDn, bp)
    [hvnUp, hvnDn, lvnUp, lvnDn, atCode]

// POC over a fixed lookback (migration sampling), outlier-clipped
f_pocSimple(simple int lb, simple int nb, float kSig) =>
    float amin = na
    float amax = na
    float sum = 0.0
    float ssq = 0.0
    int cnt = 0
    float poc = na
    for k = 0 to lb - 1
        float hk = high[k]
        float lk = low[k]
        float ck = close[k]
        if not na(hk) and not na(lk) and not na(ck)
            amin := na(amin) ? lk : math.min(amin, lk)
            amax := na(amax) ? hk : math.max(amax, hk)
            float src = (hk + lk + ck) / 3.0
            sum := sum + src
            ssq := ssq + src * src
            cnt := cnt + 1
    if cnt > 0 and not na(amin) and not na(amax) and amax > amin
        float mean = sum / cnt
        float vrc = math.max(0.0, ssq / cnt - mean * mean)
        float sd = math.sqrt(vrc)
        float loB = math.max(amin, mean - kSig * sd)
        float hiB = math.min(amax, mean + kSig * sd)
        if hiB <= loB
            loB := amin
            hiB := amax
        float bs = (hiB - loB) / nb
        arrV = array.new_float(nb, 0.0)
        for k = 0 to lb - 1
            float hk = high[k]
            float lk = low[k]
            float ck = close[k]
            float vk = volume[k]
            if not na(hk) and not na(lk) and not na(ck) and not na(vk)
                float src = (hk + lk + ck) / 3.0
                if src >= loB and src <= hiB
                    int bi = int((src - loB) / bs)
                    bi := bi < 0 ? 0 : (bi > nb - 1 ? nb - 1 : bi)
                    array.set(arrV, bi, array.get(arrV, bi) + vk)
        int pb = 0
        float pv = -1.0
        for i = 0 to nb - 1
            float v = array.get(arrV, i)
            if v > pv
                pv := v
                pb := i
        poc := loB + (pb + 0.5) * bs
    poc

    if hasShort and not na(eS) and not na(tS) and not na(sS)
        if showForecast
            array.push(flines, line.new(b0, eS, b1, tS, xloc=xloc.bar_index, color=colShort, width=2))
            array.push(flines, line.new(b0, eS, b1, sS, xloc=xloc.bar_index, color=color.new(colShort, 45), width=1, style=line.style_dashed))
            made += 2
        array.push(pgEntry, eS)
        array.push(pgStop, sS)
        array.push(pgTgt, tS)
        array.push(pgDir, -1)
    if made > 0
        array.push(fcounts, made)
        while array.size(fcounts) > fcKeep
            int c = array.shift(fcounts)
            for _i = 1 to c
                if array.size(flines) > 0
                    line.delete(array.shift(flines))
    lastKey := fcKey
    while array.size(pgEntry) > maxPending
        array.shift(pgEntry)
        array.shift(pgStop)
        array.shift(pgTgt)
        array.shift(pgDir)

// ============================ PLAN VERDICTS (graded at entry touch) ============================
bool vTake = false
bool vMarg = false
bool vSkip = false
if showVerdict and barstate.isconfirmed and array.size(pgEntry) > 0
    int idx = array.size(pgEntry) - 1
    while idx >= 0
        float pe = array.get(pgEntry, idx)
        float ps = array.get(pgStop, idx)
        float pt = array.get(pgTgt, idx)
        int   pd = array.get(pgDir, idx)
        bool touched = pd == 1 ? (low <= pe) : (high >= pe)
        if touched
            int ptouch = 0
            for j = 1 to freshW
                float lj = low[j]
                float hj = high[j]
                if not na(lj) and not na(hj) and lj <= pe + half and hj >= pe - half
                    ptouch := ptouch + 1
            bool fresh = ptouch <= freshThr
            bool lowVol = not na(volPct) and volPct < volLoThr
            bool opp = pd == 1 ? (state == "Downtrend — sell rips") : (state == "Uptrend — buy dips")
            bool aligned = pd == 1 ? hasLong : hasShort
            string tier = opp ? "SKIP" : (fresh and not lowVol) ? "TAKE" : "MARGINAL"
            color vc = tier == "TAKE" ? color.lime : tier == "SKIP" ? color.red : color.orange
            string mk = tier == "TAKE" ? "✓" : tier == "SKIP" ? "✗" : "~"
            string why = (fresh ? "fresh" : "tested") + " · " + (lowVol ? "vol lo" : "vol ok") + " · " + (opp ? "bias opp" : aligned ? "bias aligned" : "range")
            label lb = label.new(bar_index, pe, mk, xloc=xloc.bar_index, yloc=yloc.price, style=pd == 1 ? label.style_label_up : label.style_label_down, color=color.new(vc, 15), textcolor=color.white, size=size.small, tooltip=(pd == 1 ? "LONG" : "SHORT") + " entry " + f_fp(pe) + " → tgt " + f_fp(pt) + " · stop " + f_fp(ps) + "\nverdict: " + tier + "  (" + why + ")")
            array.push(vlabels, lb)
            if tier == "TAKE"
                vTake := true
            else if tier == "SKIP"
                vSkip := true
            else
                vMarg := true
            while array.size(vlabels) > keepVerdicts
                label.delete(array.shift(vlabels))
            array.remove(pgEntry, idx)
            array.remove(pgStop, idx)
            array.remove(pgTgt, idx)
            array.remove(pgDir, idx)
        idx := idx - 1
alertcondition(vTake, "Entry valid (Take)", "MTF Plan: price reached a pending entry — read still supports it")
alertcondition(vMarg, "Entry marginal", "MTF Plan: price reached a pending entry — downgraded (tested / low-vol / range)")
alertcondition(vSkip, "Entry vetoed (Skip)", "MTF Plan: price reached a pending entry — bias flipped opposite")

bool biasChg = state != state[1]
