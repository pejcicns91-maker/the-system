//@version=6
indicator("C-Bridge CB2", overlay=true, max_lines_count=500, max_labels_count=500)

// ── inputs ───────────────────────────────────────────────────────────────
payload   = input.text_area("", "CB2 payload (paste ONE asset line from the brief)")
showP85   = input.bool(false, "Show p85 budget caps")
showScen  = input.bool(true,  "Show scenario glyphs")
extBack   = input.int(30, "Level line lookback (bars)", minval=5, maxval=300)
laneWidth = input.int(12, "Glyph width (bars)", minval=6, maxval=40)
laneGap   = input.int(4,  "Gap between glyphs (bars)", minval=1, maxval=20)

// ── object registry ──────────────────────────────────────────────────────
var array<line>  LNS = array.new<line>()
var array<label> LBS = array.new<label>()
f_wipe() =>
    if array.size(LNS) > 0
        for i = 0 to array.size(LNS) - 1
            line.delete(array.get(LNS, i))
        array.clear(LNS)
    if array.size(LBS) > 0
        for i = 0 to array.size(LBS) - 1
            label.delete(array.get(LBS, i))
        array.clear(LBS)

f_hline(float y, color c, string st, int w) =>
    array.push(LNS, line.new(bar_index - extBack, y, bar_index, y, extend=extend.right, color=c, style=st, width=w))

f_seg(int x1, float y1, int x2, float y2, color c, string st, int w) =>
    array.push(LNS, line.new(x1, y1, x2, y2, extend=extend.none, color=c, style=st, width=w))

f_levlab(float y, string txt, color c, int slot) =>
    array.push(LBS, label.new(bar_index + 1 + slot * 3, y, txt, style=label.style_label_left, color=color.new(c, 88), textcolor=c, size=size.small))

// ── payload helpers ──────────────────────────────────────────────────────
f_val(array<string> parts, string key) =>
    string out = na
    if array.size(parts) > 0
        for i = 0 to array.size(parts) - 1
            p = array.get(parts, i)
            if na(out) and str.startswith(p, key + ":")
                out := str.substring(p, str.length(key) + 1)
    out

f_num(array<string> parts, string key) =>
    str.tonumber(f_val(parts, key))

// ── info table ───────────────────────────────────────────────────────────
var table T = table.new(position.top_right, 1, 14, bgcolor=color.new(color.black, 20), border_width=1)
f_row(int r, string txt, color c) =>
    table.cell(T, 0, r, txt, text_color=c, text_halign=text.align_left, text_size=size.small)

// ── main ─────────────────────────────────────────────────────────────────
if barstate.islast
    f_wipe()
    if not str.startswith(payload, "CB2|")
        f_row(0, "C-Bridge: paste a CB2 payload line", color.orange)
    else
        parts = str.split(payload, "|")
        aName = array.size(parts) > 1 ? array.get(parts, 1) : "?"
        aDate = array.size(parts) > 2 ? array.get(parts, 2) : "?"
        U    = f_num(parts, "U"),    p85 = f_num(parts, "p85"), p70 = f_num(parts, "p70")
        dtyp = f_val(parts, "type")
        ddir = f_val(parts, "dir"),  dstr = f_val(parts, "str"), bas = f_val(parts, "basis")
        PDH = f_num(parts, "PDH"), PDL = f_num(parts, "PDL"), PDC = f_num(parts, "PDC")
        ONH = f_num(parts, "ONH"), ONL = f_num(parts, "ONL"), RND = f_num(parts, "RND")
        S08 = f_num(parts, "S08")
        ob  = f_val(parts, "ob"),  ovl = f_val(parts, "ovl"), fomc = f_val(parts, "fomc")

        // 08:00 / 14:00 anchors from the payload date (exact on 24/7 charts)
        int x8 = na
        int x14 = na
        ds = str.split(aDate, "-")
        if array.size(ds) == 3
            yy = int(str.tonumber(array.get(ds, 0)))
            mm = int(str.tonumber(array.get(ds, 1)))
            dd = int(str.tonumber(array.get(ds, 2)))
            msbar = timeframe.in_seconds() * 1000
            t8  = timestamp("America/New_York", yy, mm, dd, 8, 0)
            t14 = timestamp("America/New_York", yy, mm, dd, 14, 0)
            x8  := bar_index - math.round((time - t8)  / msbar)
            x14 := bar_index - math.round((time - t14) / msbar)
            x8  := math.min(x8,  bar_index + 450)
            x14 := math.min(x14, bar_index + 480)

        // level map
        cLev = color.new(color.gray, 45)
        if not na(PDH)
            f_hline(PDH, cLev, line.style_solid, 1),  f_levlab(PDH, "PDH", color.gray, 0)
        if not na(PDL)
            f_hline(PDL, cLev, line.style_solid, 1),  f_levlab(PDL, "PDL", color.gray, 1)
        if not na(PDC)
            f_hline(PDC, color.new(color.silver, 25), line.style_solid, 1), f_levlab(PDC, "PDC", color.silver, 0)
        if not na(ONH)
            f_hline(ONH, color.new(color.blue, 40),   line.style_solid, 1), f_levlab(ONH, "ONH", color.blue, 1)
        if not na(ONL)
            f_hline(ONL, color.new(color.blue, 40),   line.style_solid, 1), f_levlab(ONL, "ONL", color.blue, 0)
        if not na(RND)
            f_hline(RND, color.new(color.yellow, 55), line.style_dotted, 1), f_levlab(RND, "RND", color.yellow, 1)
        if not na(S08)
            f_hline(S08, color.new(color.white, 55),  line.style_dotted, 1), f_levlab(S08, "S08", color.white, 0)

        // budget caps + window guides
        float yLo = na
        float yHi = na
        if not na(S08) and not na(U) and not na(PDC)
            capU = 0.70 * U / 100.0 * PDC
            f_hline(S08 + capU, color.new(color.purple, 35), line.style_solid, 1), f_levlab(S08 + capU, "U+", color.purple, 1)
            f_hline(S08 - capU, color.new(color.purple, 35), line.style_solid, 1), f_levlab(S08 - capU, "U-", color.purple, 0)
            if showP85 and not na(p85)
                cap85 = 0.70 * p85 / 100.0 * PDC
                f_hline(S08 + cap85, color.new(color.purple, 70), line.style_dashed, 1)
                f_hline(S08 - cap85, color.new(color.purple, 70), line.style_dashed, 1)
            yLo := S08 - capU * 1.25
            yHi := S08 + capU * 1.25
        if not na(x8) and not na(yLo)
            f_seg(x8,  yLo, x8,  yHi, color.new(color.gray, 30), line.style_dotted, 1)
            array.push(LBS, label.new(x8, yLo, "08:00", style=label.style_label_up, color=color.new(color.gray, 90), textcolor=color.gray, size=size.small))
        if not na(x14) and not na(yLo)
            f_seg(x14, yLo, x14, yHi, color.new(color.gray, 30), line.style_dotted, 1)
            array.push(LBS, label.new(x14, yLo, "14:00", style=label.style_label_up, color=color.new(color.gray, 90), textcolor=color.gray, size=size.small))

        // scenario glyphs anchored at 08:00
        scTxt = ""
        if showScen and array.size(parts) > 0
            xA = na(x8) ? bar_index + 6 : x8 + 1
            scN = 0
            for i = 0 to array.size(parts) - 1
                p = array.get(parts, i)
                if str.startswith(p, "SC")
                    seg = str.substring(p, str.pos(p, ":") + 1)
                    f = str.split(seg, ",")
                    if array.size(f) >= 5
                        k    = array.get(f, 0)
                        trig = str.tonumber(array.get(f, 1))
                        tgt  = str.tonumber(array.get(f, 2))
                        inv  = str.tonumber(array.get(f, 3))
                        tag  = array.get(f, 4)
                        scN += 1
                        kName = k == "L" ? "LONG" : k == "S" ? "SHORT" : "FADE"
                        base  = k == "L" ? color.lime : k == "S" ? color.red : color.orange
                        c     = tag == "lean" ? base : color.new(base, 30)
                        x0 = xA + (scN - 1) * (laneWidth + laneGap)
                        x1 = x0 + int(laneWidth * 0.45)
                        x2 = x0 + laneWidth
                        if not na(trig)
                            f_seg(x0, trig, x1, trig, c, line.style_solid, 3)
                            array.push(LBS, label.new(x0, trig, "S" + str.tostring(scN) + " " + kName + (tag == "lean" ? " ★" : ""), style=k == "S" ? label.style_label_lower_right : label.style_label_upper_right, color=color.new(base, 80), textcolor=c, size=size.small))
                        if not na(trig) and not na(tgt)
                            f_seg(x1, trig, x2, tgt, c, line.style_dashed, 2)
                            f_seg(x2, tgt, x2 + 2, tgt, c, line.style_dashed, 2)
                        if not na(trig) and not na(inv)
                            f_seg(x1, trig, x2, inv, c, line.style_dotted, 1)
                            f_seg(x2, inv, x2 + 2, inv, c, line.style_dotted, 1)
                        scTxt += "S" + str.tostring(scN) + " " + kName + " " + str.tostring(trig) + " > " + str.tostring(tgt) + " x " + str.tostring(inv) + " [" + tag + "]\n"

        // panel
        today = str.format_time(timenow, "yyyy-MM-dd", "America/New_York")
        stale = aDate != today
        f_row(0, "C-BRIDGE  " + aName + "  " + aDate + (stale ? "  (STALE!)" : ""), stale ? color.orange : color.aqua)
        f_row(1, "range U " + str.tostring(U) + "%   p70 " + str.tostring(p70) + "%   p85 " + str.tostring(p85) + "%", color.white)
        f_row(2, "day type: " + dtyp, color.white)
        f_row(3, "dir: " + ddir + " (" + dstr + ")  [" + bas + "]", ddir == "up" ? color.lime : ddir == "down" ? color.red : color.gray)
        f_row(4, "Option B: " + ob + (ovl == "1" ? "   !! OVERLAP" : ""), ovl == "1" ? color.red : color.white)
        f_row(5, "arm: 15m CLOSE beyond trigger | dies: close beyond inval, or 14:00", color.new(color.silver, 20))
        if fomc == "1"
            f_row(6, "FOMC DAY", color.orange)
        f_row(7, scTxt == "" ? "" : scTxt, color.silver)
